package com.zara.assistant.tts

import android.content.Context
import android.speech.tts.TextToSpeech
import android.util.Log
import java.util.Locale

class ZaraTts(context: Context) : TextToSpeech.OnInitListener {

    companion object {
        private const val TAG = "ZaraTts"
        private const val UTTERANCE_ID = "zara_utterance"
    }

    interface Callback {
        fun onSpeechStarted()
        fun onSpeechDone()
    }

    private var tts: TextToSpeech = TextToSpeech(context.applicationContext, this)
    private var callback: Callback? = null
    @Volatile private var ready = false

    fun setCallback(cb: Callback?) { callback = cb }

    override fun onInit(status: Int) {
        if (status == TextToSpeech.SUCCESS) {
            val result = tts.setLanguage(Locale.US)
            if (result == TextToSpeech.LANG_MISSING_DATA || result == TextToSpeech.LANG_NOT_SUPPORTED) {
                tts.setLanguage(Locale.getDefault())
            }
            applyFemaleVoice()
            tts.setOnUtteranceProgressListener(object : android.speech.tts.UtteranceProgressListener() {
                override fun onStart(utteranceId: String?) { callback?.onSpeechStarted() }
                override fun onDone(utteranceId: String?) { callback?.onSpeechDone() }
                @Deprecated("Deprecated in API")
                override fun onError(utteranceId: String?) { callback?.onSpeechDone() }
            })
            ready = true
        }
    }

    private fun applyFemaleVoice() {
        try {
            val voices = tts.voices ?: return
            val female = voices.firstOrNull { v ->
                v.locale?.language == "en" &&
                    (v.name.contains("female", ignoreCase = true) ||
                        v.name.contains("woman", ignoreCase = true) ||
                        v.name.contains("zira", ignoreCase = true) ||
                        v.name.contains("samantha", ignoreCase = true) ||
                        v.name.contains("google uk english female", ignoreCase = true))
            } ?: voices.firstOrNull { v -> v.locale?.language == "en" }
            if (female != null) tts.voice = female
        } catch (e: Exception) { }
        tts.setPitch(1.1f)
        tts.setSpeechRate(0.95f)
    }

    fun speak(text: String) {
        if (!ready || text.isBlank()) return
        tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, UTTERANCE_ID)
    }

    fun stop() { tts.stop() }
    fun shutdown() { tts.stop(); tts.shutdown(); ready = false }
    fun isReady(): Boolean = ready
}
com/zara/assistant/stt/ZaraSpeechRecognizer.kt

package com.zara.assistant.stt

import android.content.Context
import android.content.Intent
import android.os.Bundle
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import android.util.Log
import java.util.Locale

class ZaraSpeechRecognizer(private val context: Context) {

    companion object { private const val TAG = "ZaraSTT" }

    interface Listener {
        fun onReady()
        fun onPartial(text: String)
        fun onResult(text: String)
        fun onError(code: Int)
        fun onEnd()
    }

    private var recognizer: SpeechRecognizer? = null
    private var listener: Listener? = null

    fun setListener(l: Listener) { listener = l }

    fun startListening() {
        if (!SpeechRecognizer.isRecognitionAvailable(context)) {
            listener?.onError(SpeechRecognizer.ERROR_SERVICE_UNAVAILABLE)
            return
        }
        stopListening()
        recognizer = SpeechRecognizer.createSpeechRecognizer(context).apply {
            setRecognitionListener(object : RecognitionListener {
                override fun onReadyForSpeech(params: Bundle?) { listener?.onReady() }
                override fun onBeginningOfSpeech() {}
                override fun onRmsChanged(rmsdB: Float) {}
                override fun onBufferReceived(buffer: ByteArray?) {}
                override fun onEndOfSpeech() { listener?.onEnd() }
                override fun onError(error: Int) { listener?.onError(error) }
                override fun onResults(results: Bundle?) {
                    val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                    listener?.onResult(matches?.firstOrNull().orEmpty())
                }
                override fun onPartialResults(partialResults: Bundle?) {
                    val partial = partialResults
                        ?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                        ?.firstOrNull().orEmpty()
                    if (partial.isNotEmpty()) listener?.onPartial(partial)
                }
                override fun onEvent(eventType: Int, params: Bundle?) {}
            })
        }
        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
            putExtra(RecognizerIntent.EXTRA_LANGUAGE, Locale.getDefault())
            putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)
            putExtra("android.speech.extra.DICTATION_MODE", true)
        }
        recognizer?.startListening(intent)
    }

    fun stopListening() {
        recognizer?.stopListening()
        recognizer?.destroy()
        recognizer = null
    }

    fun destroy() { stopListening() }
}
com/zara/assistant/net/NetworkUtil.kt

package com.zara.assistant.net

import android.content.Context
import android.net.ConnectivityManager
import android.net.NetworkCapabilities

object NetworkUtil {
    fun isOnline(context: Context): Boolean {
        val cm = context.getSystemService(Context.CONNECTIVITY_SERVICE) as? ConnectivityManager
            ?: return false
        val network = cm.activeNetwork ?: return false
        val caps = cm.getNetworkCapabilities(network) ?: return false
        return caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET) &&
            (caps.hasTransport(NetworkCapabilities.TRANSPORT_WIFI) ||
                caps.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR) ||
                caps.hasTransport(NetworkCapabilities.TRANSPORT_ETHERNET))
    }
}

package com.zara.assistant.ai

import android.util.Log
import com.zara.assistant.net.NetworkUtil
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONArray
import org.json.JSONObject
import java.util.concurrent.TimeUnit

class OnlineAiHandler(private val context: android.content.Context) : AiHandler {

    companion object {
        private const val TAG = "OnlineAi"
        private const val API_URL = "https://api.openai.com/v1/chat/completions"
        private const val API_KEY = "REPLACE_WITH_YOUR_API_KEY"
        private const val MODEL = "gpt-3.5-turbo"
        private const val SYSTEM_PROMPT =
            "You are Zara (ज़ारा), a helpful, friendly female voice assistant. " +
            "Keep answers short and conversational since they will be spoken aloud."
    }

    private val client = OkHttpClient.Builder()
        .connectTimeout(15, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .build()

    private val jsonMedia = "application/json; charset=utf-8".toMediaType()

    override fun isAvailable(): Boolean = NetworkUtil.isOnline(context)

    override fun getReply(input: String, callback: (String) -> Unit) {
        if (!isAvailable() || API_KEY == "REPLACE_WITH_YOUR_API_KEY") {
            callback("")
            return
        }
        try {
            val body = JSONObject().apply {
                put("model", MODEL)
                val messages = JSONArray()
                messages.put(JSONObject().put("role", "system").put("content", SYSTEM_PROMPT))
                messages.put(JSONObject().put("role", "user").put("content", input))
                put("messages", messages)
                put("max_tokens", 300)
            }.toString()

            val request = Request.Builder()
                .url(API_URL)
                .header("Authorization", "Bearer $API_KEY")
                .header("Content-Type", "application/json")
                .post(body.toRequestBody(jsonMedia))
                .build()

            client.newCall(request).execute().use { response ->
                if (!response.isSuccessful) { callback(""); return }
                val reply = parseReply(response.body?.string().orEmpty())
                callback(reply)
            }
        } catch (e: Exception) {
            Log.e(TAG, "Online AI request failed", e)
            callback("")
        }
    }

    private fun parseReply(json: String): String = try {
        JSONObject(json).getJSONArray("choices")
            .getJSONObject(0).getJSONObject("message").getString("content").trim()
    } catch (e: Exception) { "" }
}
com/zara/assistant/ai/OfflineResponseHandler.kt

package com.zara.assistant.ai

import com.zara.assistant.actions.AppLauncher
import com.zara.assistant.actions.CommandType

class OfflineResponseHandler : AiHandler {

    private val greetings = listOf("Hi, I'm Zara. How can I help?", "Hello! Zara here.", "Hey there, what can I do for you?")
    private val farewells = listOf("Goodbye! Talk soon.", "See you later!", "Bye for now.")

    override fun isAvailable(): Boolean = true

    override fun getReply(input: String, callback: (String) -> Unit) {
        callback(generate(input))
    }

    fun generate(input: String): String {
        val text = input.trim().lowercase()
        if (text.isBlank()) return "I didn't catch that. Could you say it again?"
        if (text.contains(Regex("\\b(hi|hello|hey|namaste|namaskar)\\b"))) return greetings.random()
        if (text.contains(Regex("\\b(bye|goodbye|see you|good night)\\b"))) return farewells.random()
        if (text.contains("how are you")) return "I'm doing great, thanks for asking! What would you like me to do?"
        if (text.contains(Regex("\\b(who are you|your name|what are you)\\b")))
            return "I'm Zara, your offline voice assistant. I can open apps, scroll reels, and chat when you're online."
        if (text.contains(Regex("\\b(time|what time)\\b")))
            return "It's ${java.text.SimpleDateFormat("h:mm a", java.util.Locale.getDefault()).format(java.util.Date())}."
        if (text.contains(Regex("\\b(date|what day|today)\\b")))
            return "Today is ${java.text.SimpleDateFormat("EEEE, MMMM d", java.util.Locale.getDefault()).format(java.util.Date())}."
        if (text.startsWith("open ") || text.startsWith("launch ")) {
            val app = text.substringAfter("open ").substringAfter("launch ").trim()
            return if (AppLauncher.resolvePackage(app) != null) "Opening $app." else "I don't know how to open $app yet."
        }
        if (text.contains("scroll") && text.contains("reel")) return "Scrolling reels for you."
        if (text.contains("scroll up")) return "Scrolling up."
        if (text.contains("scroll down") || text.contains("scroll")) return "Scrolling down."
        if (text == "stop" || text.contains("stop scrolling")) return "Okay, stopped."
        if (text.contains("go back") || text == "back") return "Going back."
        if (text.contains(Regex("\\b(thanks|thank you)\\b"))) return "You're welcome!"
        return "I'm offline right now, so I can only handle basic commands. " +
            "Try saying \"open YouTube\", \"scroll reels\", or ask me the time."
    }

    fun classify(text: String): CommandType {
        val t = text.trim().lowercase()
        return when {
            t.startsWith("open ") || t.startsWith("launch ") -> CommandType.OPEN_APP
            t.contains("scroll") && t.contains("reel") -> CommandType.SCROLL_REELS
            t.contains("scroll up") -> CommandType.SCROLL_UP
            t.contains("scroll down") || t == "scroll" -> CommandType.SCROLL_DOWN
            t == "stop" || t.contains("stop scrolling") -> CommandType.STOP
            t.contains("go back") || t == "back" -> CommandType.BACK
            else -> CommandType.CHAT
        }
    }
}

    )

    fun launch(context: Context, name: String): Boolean {
        val pkg = resolvePackage(name) ?: return false
        return launchPackage(context, pkg)
    }

    fun launchPackage(context: Context, pkg: String): Boolean = try {
        val intent = context.packageManager.getLaunchIntentForPackage(pkg)
        if (intent != null) {
            intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            context.startActivity(intent)
            true
        } else {
            openPlayStore(context, pkg)
            false
        }
    } catch (e: Exception) { false }

    private fun openPlayStore(context: Context, pkg: String) {
        try {
            context.startActivity(
                Intent(Intent.ACTION_VIEW, Uri.parse("market://details?id=$pkg"))
                    .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            )
        } catch (e: Exception) {
            context.startActivity(
                Intent(Intent.ACTION_VIEW, Uri.parse("https://play.google.com/store/apps/details?id=$pkg"))
                    .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            )
        }
    }

package com.zara.assistant.actions

import com.zara.assistant.ai.AiHandler
import com.zara.assistant.ai.OfflineResponseHandler

class CommandProcessor(
    private val context: android.content.Context,
    private val onlineHandler: AiHandler,
    private val offlineHandler: OfflineResponseHandler
) {
    interface Callback { fun onReply(reply: String) }

    fun process(text: String, callback: Callback) {
        when (offlineHandler.classify(text)) {
            CommandType.OPEN_APP -> {
                val appName = text.substringAfter("open ").substringAfter("launch ").trim()
                val launched = AppLauncher.launch(context, appName)
                callback.onReply(if (launched) "Opening $appName." else "I couldn't open $appName. It may not be installed.")
            }
            CommandType.SCROLL_REELS -> {
                val ok = AccessibilityController.startScrollingReels()
                callback.onReply(if (ok) "Scrolling reels. Say stop when you're done." else "I need the accessibility service enabled to scroll. Please turn it on in settings.")
            }
            CommandType.SCROLL_UP -> {
                val ok = AccessibilityController.scrollUp()
                callback.onReply(if (ok) "Scrolled up." else "I can't scroll right now. Enable the accessibility service.")
            }
            CommandType.SCROLL_DOWN -> {
                val ok = AccessibilityController.scrollDown()
                callback.onReply(if (ok) "Scrolled down." else "I can't scroll right now. Enable the accessibility service.")
            }
            CommandType.STOP -> { AccessibilityController.stopScrolling(); callback.onReply("Okay, stopped.") }
            CommandType.BACK -> {
                val ok = AccessibilityController.pressBack()
                callback.onReply(if (ok) "Going back." else "I can't go back right now. Enable the accessibility service.")
            }
            CommandType.CHAT -> handleChat(text, callback)
        }
    }

    private fun handleChat(text: String, callback: Callback) {
        if (onlineHandler.isAvailable()) {
            Thread {
                val latch = java.util.concurrent.CountDownLatch(1)
                var onlineReply = ""
                runCatching {
                    onlineHandler.getReply(text) { reply -> onlineReply = reply; latch.countDown() }
                }
                latch.await(35, java.util.concurrent.TimeUnit.SECONDS)
                callback.onReply(if (onlineReply.isNotBlank()) onlineReply else offlineHandler.generate(text))
            }.start()
        } else {
            callback.onReply(offlineHandler.generate(text))
        }
    }
}
com/zara/assistant/actions/AssistantOrchestrator.kt

package com.zara.assistant.actions

import android.content.Context
import android.util.Log
import com.zara.assistant.ai.OfflineResponseHandler
import com.zara.assistant.ai.OnlineAiHandler
import com.zara.assistant.stt.ZaraSpeechRecognizer
import com.zara.assistant.tts.ZaraTts

class AssistantOrchestrator(private val context: Context) {

    companion object { private const val TAG = "ZaraOrchestrator" }

    interface AssistantListener {
        fun onStateChanged(state: AssistantState)
        fun onUserMessage(text: String)
        fun onZaraMessage(text: String)
        fun onError(message: String)
    }

    enum class AssistantState { IDLE, LISTENING, THINKING, SPEAKING }

    private val tts = ZaraTts(context)
    private val stt = ZaraSpeechRecognizer(context)
    private val onlineHandler = OnlineAiHandler(context)
    private val offlineHandler = OfflineResponseHandler()
    private val processor = CommandProcessor(context, onlineHandler, offlineHandler)

    private var listener: AssistantListener? = null
    @Volatile private var state = AssistantState.IDLE
    @Volatile private var listening = false

    fun setListener(l: AssistantListener) {
        listener = l
        tts.setCallback(object : ZaraTts.Callback {
            override fun onSpeechStarted() { setState(AssistantState.SPEAKING) }
            override fun onSpeechDone() {
                setState(AssistantState.IDLE)
                if (listening) startListening()
            }
        })
        stt.setListener(object : ZaraSpeechRecognizer.Listener {
            override fun onReady() { setState(AssistantState.LISTENING) }
            override fun onPartial(text: String) {}
            override fun onResult(text: String) {
                if (text.isBlank()) {
                    setState(AssistantState.IDLE)
                    if (listening) startListening()
                    return
                }
                listener?.onUserMessage(text)
                setState(AssistantState.THINKING)
                processor.process(text, object : CommandProcessor.Callback {
                    override fun onReply(reply: String) {
                        listener?.onZaraMessage(reply)
                        tts.speak(reply)
                    }
                })
            }
            override fun onError(code: Int) {
                val msg = friendlySttError(code)
                if (msg != null) listener?.onError(msg)
                setState(AssistantState.IDLE)
                if (listening) startListening()
            }
            override fun onEnd() {}
        })
    }

    fun start() { listening = true; startListening() }

    fun stop() {
        listening = false
        stt.stopListening()
        tts.stop()
        setState(AssistantState.IDLE)
    }

    fun shutdown() { listening = false; stt.destroy(); tts.shutdown() }

    fun speak(text: String) { listener?.onZaraMessage(text); tts.speak(text) }

    private fun startListening() {
        if (!listening) return
        stt.stopListening()
        stt.startListening()
    }

    private fun setState(s: AssistantState) { state = s; listener?.onStateChanged(s) }

    private fun friendlySttError(code: Int): String? = when (code) {
        android.speech.SpeechRecognizer.ERROR_NO_MATCH -> "I didn't catch that. Could you repeat?"
        android.speech.SpeechRecognizer.ERROR_SPEECH_TIMEOUT -> "I didn't hear anything. Tap the mic when you're ready."
        android.speech.SpeechRecognizer.ERROR_AUDIO -> "There was a problem with the microphone."
        android.speech.SpeechRecognizer.ERROR_RECOGNIZER_BUSY -> "Give me a second, I'm still processing."
        else -> null
    }
}
com/zara/assistant/MyAccessibilityService.kt

package com.zara.assistant

import android.accessibilityservice.AccessibilityService
import android.accessibilityservice.GestureDescription
import android.graphics.Path
import android.os.Build
import android.util.Log
import android.view.accessibility.AccessibilityEvent
import androidx.annotation.RequiresApi
import com.zara.assistant.actions.AccessibilityController
import kotlin.concurrent.thread

class MyAccessibilityService : AccessibilityService() {

    companion object {
        private const val TAG = "ZaraA11y"
        private const val SCROLL_PAUSE_MS = 4000L
    }

    @Volatile private var scrollingReels = false

    override fun onServiceConnected() {
        super.onServiceConnected()
        AccessibilityController.attach(this)
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        if (event == null) return
        Log.v(TAG, "event: type=0x${event.eventType.toString(16)} pkg=${event.packageName}")
    }

    override fun onInterrupt() { stopScrolling() }

    override fun onUnbind(intent: android.content.Intent?): Boolean {
        AccessibilityController.detach()
        return super.onUnbind(intent)
    }

    fun scrollDown(): Boolean = dispatchSwipe(startY = 0.70f, endY = 0.20f)
    fun scrollUp(): Boolean = dispatchSwipe(startY = 0.20f, endY = 0.70f)

    fun startScrollingReels() {
        if (scrollingReels) return
        scrollingReels = true
        thread(name = "zara-reel-scroll") {
            while (scrollingReels) {
                dispatchSwipe(startY = 0.75f, endY = 0.20f)
                try { Thread.sleep(SCROLL_PAUSE_MS) } catch (_: InterruptedException) { break }
            }
        }
    }

    fun stopScrolling() { scrollingReels = false }

    fun pressBack(): Boolean = performGlobalAction(GLOBAL_ACTION_BACK)

    @RequiresApi(Build.VERSION_CODES.N)
    private fun dispatchSwipe(startY: Float, endY: Float): Boolean = try {
        val w = resources.displayMetrics.widthPixels
        val h = resources.displayMetrics.heightPixels
        val centerX = w / 2f
        val path = Path().apply {
            moveTo(centerX, h * startY)
            lineTo(centerX, h * endY)
        }
        val stroke = GestureDescription.StrokeDescription(path, 0L, 400L)
        val gesture = GestureDescription.Builder().addStroke(stroke).build()
        dispatchGesture(gesture, object : GestureResultCallback() {
            override fun onCompleted(gesture: GestureDescription?) {}
            override fun onCancelled(gesture: GestureDescription?) {}
        }, null)
        true
    } catch (e: Exception) { false }
}

class MainActivity : AppCompatActivity(), AssistantOrchestrator.AssistantListener {

    companion object { private const val REQ_RECORD_AUDIO = 101 }

    private lateinit var zara: AssistantOrchestrator
    private lateinit var micButton: ImageButton
    private lateinit var statusText: TextView
    private lateinit var modePill: TextView
    private lateinit var transcript: LinearLayout
    private lateinit var transcriptScroll: ScrollView
    private lateinit var accessibilityHint: TextView
    @Volatile private var active = false

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        micButton = findViewById(R.id.micButton)
        statusText = findViewById(R.id.statusText)
        modePill = findViewById(R.id.modePill)
        transcript = findViewById(R.id.transcript)
        transcriptScroll = findViewById(R.id.transcriptScroll)
        accessibilityHint = findViewById(R.id.accessibilityHint)

        zara = AssistantOrchestrator(this).also { it.setListener(this) }

        micButton.setOnClickListener {
            if (!hasMicPermission()) requestMicPermission() else toggleSession()
        }
        accessibilityHint.setOnClickListener {
            startActivity(Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS))
        }
        updateModePill()
        updateAccessibilityHint()
    }

    override fun onResume() { super.onResume(); updateModePill(); updateAccessibilityHint() }
    override fun onDestroy() { super.onDestroy(); zara.shutdown() }

    private fun hasMicPermission(): Boolean =
        ContextCompat.checkSelfPermission(this, Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED

    private fun requestMicPermission() {
        ActivityCompat.requestPermissions(this, arrayOf(Manifest.permission.RECORD_AUDIO), REQ_RECORD_AUDIO)
    }

    override fun onRequestPermissionsResult(requestCode: Int, permissions: Array<out String>, grantResults: IntArray) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)
        if (requestCode == REQ_RECORD_AUDIO) {
            if (grantResults.firstOrNull() == PackageManager.PERMISSION_GRANTED) toggleSession()
            else { Toast.makeText(this, R.string.mic_permission_denied, Toast.LENGTH_LONG).show(); statusText.text = getString(R.string.mic_permission_denied) }
        }
    }

    private fun toggleSession() {
        if (active) {
            zara.stop(); active = false; statusText.text = getString(R.string.idle)
        } else {
            if (!isAccessibilityEnabled()) Toast.makeText(this, R.string.accessibility_not_enabled, Toast.LENGTH_LONG).show()
            active = true; zara.start()
        }
    }

    override fun onStateChanged(state: AssistantOrchestrator.AssistantState) {
        runOnUiThread {
            statusText.text = when (state) {
                AssistantOrchestrator.AssistantState.IDLE -> getString(R.string.idle)
                AssistantOrchestrator.AssistantState.LISTENING -> getString(R.string.listening)
                AssistantOrchestrator.AssistantState.THINKING -> getString(R.string.thinking)
                AssistantOrchestrator.AssistantState.SPEAKING -> getString(R.string.speaking)
            }
        }
    }

    override fun onUserMessage(text: String) { runOnUiThread { addBubble("You: $text", true) } }
    override fun onZaraMessage(text: String) { runOnUiThread { addBubble("Zara: $text", false) } }
    override fun onError(message: String) { runOnUiThread { Toast.makeText(this, message, Toast.LENGTH_SHORT).show() } }

    private fun addBubble(text: String, isUser: Boolean) {
        val tv = layoutInflater.inflate(R.layout.item_message, transcript, false) as TextView
        tv.text = text
        tv.setBackgroundColor(ContextCompat.getColor(this, if (isUser) R.color.surface_variant else R.color.surface))
        transcript.addView(tv)
        transcriptScroll.post { transcriptScroll.fullScroll(ScrollView.FOCUS_DOWN) }
    }

    private fun updateModePill() {
        val online = NetworkUtil.isOnline(this)
        modePill.text = getString(if (online) R.string.online_mode else R.string.offline_mode)
        modePill.setTextColor(ContextCompat.getColor(this, if (online) R.color.pulse_online else R.color.pulse_offline))
    }

    private fun updateAccessibilityHint() {
        accessibilityHint.visibility = if (isAccessibilityEnabled()) View.GONE else View.VISIBLE
    }

    private fun isAccessibilityEnabled(): Boolean {
        if (AccessibilityController.isAvailable()) return true
        val enabled = Settings.Secure.getInt(contentResolver, Settings.Secure.ACCESSIBILITY_ENABLED, 0) == 1
        if (!enabled) return false
        val list = Settings.Secure.getString(contentResolver, Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES) ?: return false
        val target = packageName + "/" + MyAccessibilityService::class.java.name
        val splitter = TextUtils.SimpleStringSplitter(':').apply { setString(list) }
        while (splitter.hasNext()) { if (splitter.next().equals(target, ignoreCase = true)) return true }
        return false
    }
